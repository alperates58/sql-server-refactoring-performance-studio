CREATE VIEW dbo.AA_URETIM_MALZEME_PLANLAMA AS
SELECT
    a.[Ana İş Emri],
    a.[İş Kodu],
    a.[Sipariş No],
    a.[Ürün Kodu],
    ISNULL(i.[Kalan Miktar], 0) AS [Kalan Miktar],
    u.upl_kodu AS [Tüketim Kod],
    s.sto_isim AS [Stok Adı],
    ISNULL(sh.ToplamGiris, 0) - ISNULL(sh.ToplamCikis, 0) AS [Mevcut Stok]
FROM dbo.ISEMIRLERI a
LEFT JOIN dbo.AA_ISEMRI_MALZEME_DURUMLARI i ON i.[İş Kodu] = a.[İş Kodu]
LEFT JOIN dbo.URUN_RECETELERI u ON u.upl_ana_kod = a.[Ürün Kodu]
LEFT JOIN dbo.STOKLAR s ON s.sto_kod = u.upl_kodu
OUTER APPLY (
    SELECT
        SUM(CASE WHEN sth_tip = 0 THEN sth_miktar ELSE 0 END) AS ToplamGiris,
        SUM(CASE WHEN sth_tip = 1 THEN sth_miktar ELSE 0 END) AS ToplamCikis
    FROM dbo.STOK_HAREKETLERI
    WHERE sth_stok_kod = u.upl_kodu
) sh;
